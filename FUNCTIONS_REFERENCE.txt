PYTHON STANDARD LIBRARY

os.path.join
Combines folder and file names into one valid path.
Syntax: os.path.join(folder, file_name)
Example: os.path.join(PROJECT_ROOT, "MERGED DATASETS")

os.path.exists
Checks whether a file or folder exists.
Syntax: os.path.exists(path)
Example: os.path.exists("data/file.csv")

os.makedirs
Creates a folder, including missing parent folders.
Syntax: os.makedirs(path, exist_ok=True)
Example: os.makedirs(OUT_DIR, exist_ok=True)

open
Opens a file for reading or writing.
Syntax: open(file_path, mode, encoding="utf-8")
Example: open("data.json", "r", encoding="utf-8")

json.load
Reads JSON data from an opened file.
Syntax: json.load(file_object)
Example: json.load(notebook_file)

re.match
Checks whether text matches a pattern from its beginning.
Syntax: re.match(pattern, text)
Example: re.match(r"\s*#\s*(\d+)", source)

re.search
Finds a pattern anywhere in text.
Syntax: re.search(pattern, text)
Example: re.search(r"(^|\n)\s*##\s+", text)

re.sub
Replaces text matching a regular expression.
Syntax: re.sub(pattern, replacement, text)
Example: re.sub(r"\s+", " ", value)

warnings.filterwarnings
Controls which warning messages Python displays.
Syntax: warnings.filterwarnings(action)
Example: warnings.filterwarnings("ignore")


PANDAS

pd.read_csv
Reads a CSV file into a DataFrame.
Syntax: pd.read_csv(file_path)
Example: pd.read_csv("MERGED DATASETS/merged_crime_data.csv")

pd.to_datetime
Converts text values into dates and times.
Syntax: pd.to_datetime(values, errors="coerce")
Example: pd.to_datetime(txn["TransactionDate"], errors="coerce")

pd.to_numeric
Converts values into numbers.
Syntax: pd.to_numeric(values, errors="coerce")
Example: pd.to_numeric(txn["TransactionAmount_INR"], errors="coerce")

DataFrame.head
Returns the first rows of a DataFrame.
Syntax: dataframe.head(number)
Example: df.head(5)

DataFrame.sample
Returns random rows from a DataFrame.
Syntax: dataframe.sample(n=number, random_state=number)
Example: txn.sample(n=10000, random_state=42)

DataFrame.copy
Creates an independent copy of a DataFrame.
Syntax: dataframe.copy()
Example: banks_clean = banks[columns].copy()

DataFrame.rename
Changes column names or index names.
Syntax: dataframe.rename(columns={old_name: new_name})
Example: txn.rename(columns={"TransactionAmount (INR)": "TransactionAmount_INR"})

DataFrame.drop
Removes columns, rows, or duplicates from a DataFrame.
Syntax: dataframe.drop(columns=[column_name])
Example: dataframe.drop(columns=["_join_key"])

DataFrame.drop_duplicates
Removes repeated rows.
Syntax: dataframe.drop_duplicates(inplace=True)
Example: txn.drop_duplicates(inplace=True)

DataFrame.dropna
Removes rows or columns containing missing values.
Syntax: dataframe.dropna(subset=[column_name])
Example: txn.dropna(subset=["TransactionID", "CustomerID"])

DataFrame.fillna
Replaces missing values with a specified value.
Syntax: dataframe.fillna(value)
Example: rbi["ATMs_On_Site"].fillna(0)

DataFrame.astype
Converts a column or DataFrame to a specified data type.
Syntax: dataframe.astype(data_type)
Example: df["Year"].astype("Int64")

DataFrame.str.strip
Removes whitespace from the beginning and end of text.
Syntax: dataframe["column"].str.strip()
Example: banks["Bank"].str.strip()

DataFrame.str.upper
Converts text to uppercase.
Syntax: dataframe["column"].str.upper()
Example: txn["CustGender"].str.upper()

DataFrame.str.contains
Checks whether text contains a pattern.
Syntax: dataframe["column"].str.contains(pattern, na=True)
Example: xls["State/UT"].str.contains(r"TOTAL|ALL INDIA", na=True)

DataFrame.groupby
Groups rows by one or more columns for aggregation.
Syntax: dataframe.groupby(columns)
Example: crime.groupby("State_UT")

GroupBy.agg
Calculates one or more summary values for grouped data.
Syntax: grouped.agg(new_column=(column, function))
Example: rbi.groupby("Bank_Name").agg(ATMs_Total=("ATMs_Total", "mean"))

GroupBy.sum
Calculates totals for each group.
Syntax: dataframe.groupby(column)[value_column].sum()
Example: crime.groupby("State_UT")["Total_Crimes_Reported"].sum()

GroupBy.size
Counts rows in each group.
Syntax: dataframe.groupby(columns).size()
Example: txn.groupby(["DayOfWeek", "TransactionHour"]).size()

DataFrame.merge
Joins two DataFrames using matching columns.
Syntax: left.merge(right, on=column, how="left")
Example: banks_clean.merge(rbi_agg, on="_join_key", how="left")

DataFrame.sort_values
Sorts rows by one or more columns.
Syntax: dataframe.sort_values(columns)
Example: df.sort_values(["Bank_Name", "Year"])

DataFrame.reset_index
Creates a new sequential index.
Syntax: dataframe.reset_index(drop=True)
Example: df.reset_index(drop=True)

DataFrame.melt
Converts wide data into long data.
Syntax: dataframe.melt(id_vars=column, var_name=name, value_name=name)
Example: trends.melt(id_vars="Year", var_name="Metric", value_name="Value")

DataFrame.pivot
Reshapes long data into a table.
Syntax: dataframe.pivot(index=column, columns=column, values=column)
Example: pivot_hw.pivot(index="DayOfWeek", columns="TransactionHour", values="Count")

DataFrame.pivot_table
Creates a summarized pivot table.
Syntax: dataframe.pivot_table(index=column, values=column, aggfunc=function)
Example: df.pivot_table(index="Year", values="Net_Profit_Cr", aggfunc="sum")

DataFrame.nlargest
Returns rows with the largest values.
Syntax: dataframe.nlargest(number, column)
Example: latest.nlargest(15, "Total_Assets_Cr")

Series.value_counts
Counts the frequency of each unique value.
Syntax: dataframe["column"].value_counts()
Example: txn["CustLocation"].value_counts().head(10)

Series.clip
Limits values to a minimum or maximum.
Syntax: series.clip(upper=maximum)
Example: txn["TransactionAmount_INR"].clip(upper=100000)

Series.nlargest
Returns the largest values in a Series.
Syntax: series.nlargest(number)
Example: cyber_fraud.nlargest(20)

DataFrame.to_csv
Writes a DataFrame to a CSV file.
Syntax: dataframe.to_csv(file_path, index=False)
Example: merged_banks.to_csv(out1, index=False)
